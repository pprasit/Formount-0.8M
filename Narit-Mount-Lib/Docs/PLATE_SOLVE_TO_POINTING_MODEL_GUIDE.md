private void ProcessPlatesolveResult(
    CalibrationPoint point, 
    PlateSolveResult result)
{
    // 3.1.1 Mount ????????????????? (Expected)
    double expectedRa_hours = point.ExpectedRa_J2000_Rads * (12.0 / Math.PI);
    double expectedDec_deg = point.ExpectedDec_J2000_Rads * (180.0 / Math.PI);
    
    // 3.1.2 Plate solve ????????????????????? ? (Solved)
    double solvedRa_hours = result.SolvedRa_J2000_Hours;
    double solvedDec_deg = result.SolvedDec_J2000_Degrees;
    
    // 3.1.3 ????? Error = Solved - Expected
    //       (??????? mount ?????????????)
    
    // RA Error (convert to arcseconds)
    double raError_hours = solvedRa_hours - expectedRa_hours;
    double raError_arcsec = raError_hours * 15.0 * 3600.0;  // 1 hour = 15° = 54000"
    
    // Dec Error (convert to arcseconds)
    double decError_deg = solvedDec_deg - expectedDec_deg;
    double decError_arcsec = decError_deg * 3600.0;
    
    // Total Error (RMS)
    double totalError_arcsec = Math.Sqrt(
        raError_arcsec * raError_arcsec + 
        decError_arcsec * decError_arcsec
    );
    
    // 3.1.4 ?????? errors ?? CalibrationPoint
    point.SolvedRa_J2000_Rads = solvedRa_hours * (Math.PI / 12.0);
    point.SolvedDec_J2000_Rads = solvedDec_deg * (Math.PI / 180.0);
    
    point.RaError_Arcsec = raError_arcsec;
    point.DecError_Arcsec = decError_arcsec;
    point.TotalError_Arcsec = totalError_arcsec;
    
    point.NumMatchedStars = result.NumMatchedStars;
    point.RotationAngle_Deg = result.RotationAngle_Degrees;
    
    Debug.WriteLine($"[Point {point.Index}] Errors:");
    Debug.WriteLine($"  RA Error:  {raError_arcsec:F2} arcsec");
    Debug.WriteLine($"  Dec Error: {decError_arcsec:F2} arcsec");
    Debug.WriteLine($"  Total:     {totalError_arcsec:F2} arcsec");
}
