### 5. Real-time Limit Warning (UIUpdateTimer_Tick)

**New Method: CheckPositionLimits**

```csharp
private void CheckPositionLimits(double positionDeg, double velocityDeg, AzimuthAxisConfig axisConfig)
{
    double warningThreshold = axisConfig.Position_Window_Deg * 2.0; // Warning at 2x Position Window
    
    // Check CCW Limit (lower limit)
    double distanceFromCCW = positionDeg - axisConfig.CCW_Limit_Deg;
    bool ccwWarning = distanceFromCCW <= warningThreshold && distanceFromCCW > 0;
    
    if (ccwWarning && !_lastCCWWarningState)
    {
        AddLog($"[LIMIT WARNING] Approaching CCW Limit! Position: {positionDeg:F2}deg, Limit: {axisConfig.CCW_Limit_Deg:F2}deg, Distance: {distanceFromCCW:F2}deg", LogType.Error);
        _lastCCWWarningState = true;
    }
    
    // Check CW Limit (upper limit)
    double distanceFromCW = axisConfig.CW_Limit_Deg - positionDeg;
    bool cwWarning = distanceFromCW <= warningThreshold && distanceFromCW > 0;
    
    if (cwWarning && !_lastCWWarningState)
    {
        AddLog($"[LIMIT WARNING] Approaching CW Limit! Position: {positionDeg:F2}deg, Limit: {axisConfig.CW_Limit_Deg:F2}deg, Distance: {distanceFromCW:F2}deg", LogType.Error);
        _lastCWWarningState = true;
    }
    
    // Check if already exceeded limits
    if (positionDeg < axisConfig.CCW_Limit_Deg)
    {
        AddLog($"[LIMIT EXCEEDED] CCW Limit exceeded! Position: {positionDeg:F2}deg < Limit: {axisConfig.CCW_Limit_Deg:F2}deg", LogType.Error);
    }
    
    if (positionDeg > axisConfig.CW_Limit_Deg)
    {
        AddLog($"[LIMIT EXCEEDED] CW Limit exceeded! Position: {positionDeg:F2}deg > Limit: {axisConfig.CW_Limit_Deg:F2}deg", LogType.Error);
    }
}
```
